                        def computeH(im1_pts, im2_pts):
                            n = im1_pts.shape[0]
                            if n < 4:
                                raise ValueError("At least 4 point correspondences are required.")
                            
                            A = []
                            for i in range(n):
                                x, y = im1_pts[i, 0], im1_pts[i, 1]
                                xp, yp = im2_pts[i, 0], im2_pts[i, 1]
                                
                                row1 = [-x, -y, -1,  0,  0,  0,  x * xp,  y * xp,  xp]
                                row2 = [ 0,  0,  0, -x, -y, -1,  x * yp,  y * yp,  yp]
                                
                                A.append(row1)
                                A.append(row2)
                                
                            A = np.array(A, dtype=np.float64)
                            
                            _, _, Vt = np.linalg.svd(A)
                            h = Vt[-1, :]
                            
                            H = h.reshape(3, 3)
                            if H[2, 2] != 0:
                                H = H / H[2, 2]
                                
                            return H

                        def get_bounding_box(im_shape, H):
                            """
                            Predicts the bounding box of the warped image with safety clamps 
                            to prevent runaway memory errors.
                            """
                            h, w = im_shape[:2]
                            corners = np.array([
                                [0, 0, 1],
                                [w - 1, 0, 1],
                                [w - 1, h - 1, 1],
                                [0, h - 1, 1]
                            ]).T  # Shape: (3, 4)
                        
                            transformed_corners = H @ corners
                        
                            z_vals = transformed_corners[2, :]
                            z_vals[np.abs(z_vals) < 1e-5] = 1e-5
                            transformed_corners /= z_vals
                            
                            x_coords = transformed_corners[0, :]
                            y_coords = transformed_corners[1, :]
                            
                            all_x = np.append(x_coords, [0, w])
                            all_y = np.append(y_coords, [0, h])
                            
                            x_min = max(int(np.floor(all_x.min())), -2 * w)
                            x_max = min(int(np.ceil(all_x.max())), 3 * w)
                            y_min = max(int(np.floor(all_y.min())), -2 * h)
                            y_max = min(int(np.ceil(all_y.max())), 3 * h)
                            
                            return x_min, x_max, y_min, y_max
                        
                        
                        def warpImageNearestNeighbor(im, H):
                            """
                            Warps image im using homography H with Nearest Neighbor interpolation (inverse warping).
                            """
                            h, w = im.shape[:2]
                            channels = im.shape[2] if len(im.shape) == 3 else 1
                            
                            x_min, x_max, y_min, y_max = get_bounding_box(im.shape, H)
                            out_w = x_max - x_min + 1
                            out_h = y_max - y_min + 1
                            
                            x_indices = np.arange(x_min, x_max + 1)
                            y_indices = np.arange(y_min, y_max + 1)
                            X_out, Y_out = np.meshgrid(x_indices, y_indices)
                            
                            num_pixels = out_w * out_h
                            ones = np.ones((1, num_pixels))
                            coords_out = np.vstack((X_out.flatten(), Y_out.flatten(), ones))
                        
                            H_inv = np.linalg.inv(H)
                            coords_src = H_inv @ coords_out
                            coords_src /= coords_src[2, :]  
                            
                            X_src = coords_src[0, :].reshape((out_h, out_w))
                            Y_src = coords_src[1, :].reshape((out_h, out_w))
                            
                            X_nn = np.round(X_src).astype(np.int64)
                            Y_nn = np.round(Y_src).astype(np.int64)
                            
                            valid_mask = (X_nn >= 0) & (X_nn < w) & (Y_nn >= 0) & (Y_nn < h)
                            
                            X_nn_clamped = np.clip(X_nn, 0, w - 1)
                            Y_nn_clamped = np.clip(Y_nn, 0, h - 1)
                            
                            if channels > 1:
                                im_warped = np.zeros((out_h, out_w, channels), dtype=im.dtype)
                                alpha_mask = np.zeros((out_h, out_w), dtype=np.float64)
                                for c in range(channels):
                                    channel_data = im[:, :, c]
                                    sampled = channel_data[Y_nn_clamped, X_nn_clamped]
                                    sampled[~valid_mask] = 0
                                    im_warped[:, :, c] = sampled
                            else:
                                im_warped = im[Y_nn_clamped, X_nn_clamped]
                                im_warped[~valid_mask] = 0
                                im_warped = im_warped.reshape((out_h, out_w))
                                
                            alpha_mask = valid_mask.astype(np.float64)
                            
                            return im_warped, alpha_mask

I start by importing the images and converting them from the default BGR format to RGB format. After loading the images, I use pixpy to get the coordinates for the source and target images to create the homographies with and manually input them as arrays in my code.

The computeH function calculates the homography matrix taking the cordinate arrays as the input. The first check the function does is to check for the number of cordinate pairs in the arrays, if there are less than four coordinate pairs in either of the arrays, it gives an error. The function then uses Direct Linear Transformation (DLT) to create matrix A of size 2N X 9. Where each point correspondence contributes to two rows of linear equations representing Ah = 0. Then the function then factorizes the matri using the Singular Value Decomposition (SVD) and identifies the optimal least-squares solution which forms the parameters of the homography matrix. Finally, the function reshapes the vector into a 3 x 3 matrix format and scales it by its bottom-right element such that magnitude of H equals 1, before returning the final homography matrix.

                                                kernel = np.ones((9, 9), dtype=np.float32) / 81.0
                                                
                                                def convolve_with_2forloops(img, kernel):
                                                    kernel_h, kernel_w = kernel.shape
                                                    img_h, img_w = img.shape
                                                
                                                    pad_h = kernel_h // 2
                                                    pad_w = kernel_w // 2
                                                
                                                    padded_img = np.pad(img, ((pad_h, pad_h), (pad_w, pad_w)), mode = 'constant', constant_values = 0)
                                                
                                                    flip_kernel = np.flip(np.flip(kernel, axis = 0), axis = 1)
                                                
                                                    output = np.zeros_like(img, dtype=np.float32)
                                                
                                                    for i in range(img_h):
                                                        for j in range(img_w):
                                                            roi = padded_img[i : i + kernel_h, j : j + kernel_w]
                                                            output[i, j] = np.sum(roi * flip_kernel)
                                                            
                                                    return output

Section 2: Edge Detection Using Difference Operators

Now we will look at another useful way to process images and extract useful information. We are going to look at how to detect the edges of an image using the difference operators. Difference operators are uni-directional vectors that are used to extract magnitudes in the y and x directions. The dy operator works by subtracting the pixel above from the pixel below, this highlights the vertical edges as the operator moves across the image rows from top to bottom. The dx operator works in the same way but in the horizontal direction, capturing the magnitudes of the edges across the image from left to right.

The code below uses the built-in convolve function and dx and dy difference operators to capture the edges of an input image. The convolution function takes care of the padding and also makes sure that output image is the same size as the original image.

The gradient is calculated by convolving the image with an x and y difference operator respectively and calculating the sum of the square roots of the gradients in dx and dy. The following code is used to calculate the gradients.

We have first calculated the gradient magnitudes using convolved function as explained in the previous steps Then we reduce the noise in the image by implementing a binarization loop and pre-defining a list of threshold values. The threshold values are intensity cutoffs for the pixels; any pixels that are greater than or equal to the threshold value are converted to a value of 255 and any pixels below the threshold value are turned to 0. This reduces the noise by emphasizing the prominent edge features and turns the background noise features dark. Below is the sweep of the image for different threshold values.

As you can see that with lower threshold values, we get a lot of background noise from grass and other surrounding features but as we increase the threshold, the edges of the cameraman become crisper and more prominent but as we keep increasing the threshold beyond 60, we start to see the cameraman starts to turn darker and blend in with the surrounding. It is an iterative process to find the right threshold that focus on the important features, too low of a threshold and you’re looking for a needle in a haystack and too high of a threshold and you blend in with the surroundings.

The program works by first defining a column vector of size 5x1. The kernel is then multiplied by its transpose, a row vector of size 1x5 to create a symmetric 5x5 matrix with the highest value placed at its center and the sum of its elements equals to one to preserve the brightness of the original image. We then use this kernel to blur our input image. This blurred image will be used to detect edges using the DoG filters we will create next. We then calculate the DoG filters for x and y by convolving our 2D kernel with 1d vector kernels for x and y respectively. Now for comparison’s sake, we perform two different approaches here. The first approach, the two-step approach involves blurring the image first and then convolving it with the 1D difference operators dx and dy. The second, the single step approach involves convolving the original image without blurring to the 2D kernel directly. This eliminates the intermediate calculation for the blurred image and makes the program more efficient. Let’s take a look at the results.

Before we even apply binarization and reduce the image noise with threshold, you can see the gradient magnitude obtained with the DoG filter is very similar to the one obtained with two-step method where we blurred the image first. Even though the result from the two-step method and the single step method are identical, the single-step method is computationally less intensive and can process larger images quicker and more efficiently. Section 3: Image Sharpening and Unmasking Filter

So far, we have discussed convolution and how to use it to smoothen/blur images. Next, we will look at how to unblur these images using the unsharp mask filter. Since the Gaussian filter we have been using so far retains only the low frequencies from the original image, if we subtract the blurred image from the original, we are left with a sharper image with more high frequencies. The code below does exactly this, it takes an input image blurs it and then resharpens it using this technique.

Section 4: Image Blending

In this next section we will discuss more advanced image processing techniques. We will look at how to create optical illusians by combining a pair of images. By combining the low frequencies of one image to high frequencies of another we can create a visual effect that changes the image featres based on your viewing distance from the image. Since viewing from a distance reduces the visibility of sharp, high frequncy features the low frequency image dominate and when viewing the image from close up, you see more detail i.e the high frequency image.

In principal it sounds like a simple concept, simply take two images, blur the first one using a low pass gaussian filter to obtain the low frequencies from it and for second image subtract its smoothed version from the original to obtain its high frequencies, add the two together and done. In practice however, it takes some trial and error to perfectly align and scale the two together. Any misalignment and you won't get the desired effect you're looking for.

In this section we look at how to blend two images together without having a seam at he overlapping edge. if you were to take two images cut them in the middle and join left half of one to right half of the other you will get a dark unnatural seam and abrupt transition from one to the next. In order to avoid this we have to understand that imaes are like a lasagne, made up of layers, each layer contaiing different details that when all stacked on top of each other form an image. This is the technique that Burt and Adelson described in their parper. In order to achieve a smooth transition when joiing two images toether we can get a smooth natural transition if we first split the layers of the image, apply a fade to each layer depending on its frequency and then adding the layers back together to form the final image. By multiplying each layer by a different factor based on its frequency creates an even fade that gradually transitions and blurs out the image near the seam.

The code below is used to recreate the Burt Apple image from Computer Vision: Algorithms and Applications, 2nd ed. by Richard Szeliski.


                        # Laplacian Pyramid Blending (Burt & Adelson style) using Gaussian/Laplacian stacks
                        import cv2
                        import matplotlib.pyplot as plt
                        import numpy as np
                        
                        path = r"C:\Users\Muhammad Muneeb\Desktop\Documents\Masters\SYDE 671\Assignment 2\ass2 part 2 images"
                        img1 = cv2.imread(path + r"\burt_apple.png")
                        img2 = cv2.imread(path + r"\burt_orange.png")
                        
                        img1 = cv2.cvtColor(img1, cv2.COLOR_BGR2RGB).astype(np.float32) / 255.0
                        img2 = cv2.cvtColor(img2, cv2.COLOR_BGR2RGB).astype(np.float32) / 255.0
                        
                        if img1.shape != img2.shape:
                            img2 = cv2.resize(img2, (img1.shape[1], img1.shape[0]))
                        
                        h, w = img1.shape[:2]
                        
                        mask = np.zeros((h, w), dtype=np.float32)
                        mask[:, :w // 2] = 1.0
                        
                        
                        def build_gaussian_stack(image, num_levels=5, base_sigma=2.0):
                            stack = [image.copy()]
                            for i in range(1, num_levels):
                                sigma = base_sigma * (2 ** (i - 1))
                                stack.append(cv2.GaussianBlur(image, (0, 0), sigmaX=sigma, sigmaY=sigma))
                            return stack
                        
                        
                        def build_laplacian_stack(g_stack):
                            stack = [g_stack[i] - g_stack[i + 1] for i in range(len(g_stack) - 1)]
                            stack.append(g_stack[-1])
                            return stack
                        
                        
                        num_levels = 5
                        base_sigma = 2.0
                        
                        g1 = build_gaussian_stack(img1, num_levels, base_sigma)
                        g2 = build_gaussian_stack(img2, num_levels, base_sigma)
                        gm = build_gaussian_stack(mask, num_levels, base_sigma)
                        
                        l1 = build_laplacian_stack(g1)
                        l2 = build_laplacian_stack(g2)
                        
                        apple_contribs, orange_contribs, blended_stack = [], [], []
                        for a, o, m in zip(l1, l2, gm):
                            m3 = np.dstack([m, m, m])
                            ac = m3 * a
                            oc = (1.0 - m3) * o
                            apple_contribs.append(ac)
                            orange_contribs.append(oc)
                            blended_stack.append(ac + oc)
                        
                        assert np.allclose(sum(l1), img1, atol=1e-4)
                        assert np.allclose(sum(l2), img2, atol=1e-4)
                        
                        blend = np.clip(np.sum(blended_stack, axis=0), 0, 1)
                        
                        
                        def to_display(img, gain, is_residual):
                            if is_residual:
                                return np.clip(img, 0, 1)
                            return np.clip(0.5 + gain * img, 0, 1)
                        
                        
                        levels_to_show = [0, 2, 4]
                        row_labels = ["High frequency\n(level 0)", "Medium frequency\n(level 2)", "Low frequency\n(level 4)"]
                        col_labels = ["Apple (weighted)", "Orange (weighted)", "Combined"]
                        
                        fig, axes = plt.subplots(3, 3, figsize=(9, 9))
                        fig.subplots_adjust(left=0.14, right=0.99, top=0.94, bottom=0.01, wspace=0.03, hspace=0.03)
                        
                        for r, lvl in enumerate(levels_to_show):
                            is_res = (lvl == num_levels - 1)
                            peak = max(np.percentile(np.abs(x[lvl]), 99.5)
                                       for x in (apple_contribs, orange_contribs, blended_stack))
                            gain = 0.5 / max(peak, 1e-6)
                        
                            for c, stack in enumerate((apple_contribs, orange_contribs, blended_stack)):
                                ax = axes[r, c]
                                ax.imshow(to_display(stack[lvl], gain, is_res))
                                ax.set_xticks([])
                                ax.set_yticks([])
                                for spine in ax.spines.values():
                                    spine.set_visible(False)
                                if r == 0:
                                    ax.set_title(col_labels[c], fontsize=10)
                                if c == 0:
                                    ax.set_ylabel(row_labels[r], fontsize=10)
                        
                        plt.show()
                        
                        fig2, ax2 = plt.subplots(1, 3, figsize=(10, 4))
                        for ax, im, t in zip(ax2, (img1, img2, blend), ("Apple", "Orange", "Blended")):
                            ax.imshow(im)
                            ax.set_title(t, fontsize=10)
                            ax.axis("off")
                        plt.tight_layout()
                        plt.show()
